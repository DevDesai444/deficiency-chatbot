    messages: list[dict],

    temperature: float = 0.0,
    max_tokens: int = 4096,
    tool_choice: str = "auto",

    guided_model_cls: type[BaseModel] | None = None,  # D-08/Pitfall-7: source model for sanitized schema

    # This single wiring point covers ALL tool-call turns (all 7 review tools + VERDICT)
    # without requiring every caller to pass extra_body explicitly.
    #
    # PITFALL 7 (RESEARCH.md): the guided schema MUST route through tool_schema_for_databricks()
    # (via build_guided_extra_body) — NEVER the raw tool parameters dict. Raw schemas can contain
    # $ref/anyOf/pattern that vLLM backends reject. build_guided_extra_body calls
    # tool_schema_for_databricks internally and is the single mandatory sanitized builder.
    #
    # DO NOT add build_guided_extra_body_from_tools or any helper that reads raw tool["function"]["parameters"]
    # and passes it to guided decode without tool_schema_for_databricks — that is exactly Pitfall 7.
    #
    # How to supply the schema source:
    #   - For typed-tool calls (e.g. VERDICT): pass guided_model_cls=VERDICT. The sanitized

    if extra_body is None and guided_model_cls is not None and supports_guided_json(client, resolved_model):
        # Route through tool_schema_for_databricks — the mandatory sanitizer (Pitfall 7)
        extra_body = build_guided_extra_body(guided_model_cls)

    kwargs: dict = {
        "model": resolved_model,
        "messages": messages,
        "tools": tools,
        "tool_choice": tool_choice,
        "temperature": temperature,
        "max_tokens": max_tokens,

    # ... rest of retry loop unchanged ...
