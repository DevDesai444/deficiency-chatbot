def build_guided_extra_body_from_tools(tools: list[dict]) -> dict | None:
    """Build guided decode extra_body from tool schemas for chat_completion_tools.
    
    For single-tool calls: uses that tool's parameters schema directly.
    For multi-tool calls: uses the first tool's schema as the decode target.
    Returns None if tools list is empty.
    """
    from llm.reliability import build_guided_extra_body
    if not tools:

    # Extract the parameters schema from the first tool definition

    params_schema = first_tool.get("function", {}).get("parameters", {})
    if not params_schema:

    return {"structured_outputs": {"json": params_schema}}
