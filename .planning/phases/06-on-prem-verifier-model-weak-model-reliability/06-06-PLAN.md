from dataclasses import dataclass

import pytest
import structlog
from deepeval import assert_test
from deepeval.metrics import BaseMetric
from deepeval.test_case import LLMTestCase

pytestmark = pytest.mark.integration

@dataclass
class ProbeSample:
    finding_id: str
    evidence: str          # verbatim source span
    rule_citation: str     # the rule the finding cites
    failure_family: str
    expected_verdict: Literal["KEEP", "DOWNGRADE"]  # KEEP=matched-GT, DOWNGRADE=FP-tail


def load_probe_samples() -> list[ProbeSample]:
    """Load the 115 real β candidates from committed report JSONs.
    
    matched_gt_ids → KEEP-expected (5 matched-GT + MS-01 = 6 KEEP)
    all others      → DOWNGRADE-expected (~109 FP-tail)
    """
    summary = json.loads(BETA_SUMMARY_PATH.read_text())
    matched_gt_ids: set[str] = set(summary.get("matched_gt_ids", []))

    samples: list[ProbeSample] = []
    for report_path in [BETA_MVR_REPORT_PATH, BETA_MINI_REPORT_PATH]:
        if not report_path.exists():
            continue
        report = json.loads(report_path.read_text())
        for fault in report.get("faults", []):
            fid = fault.get("finding_id", fault.get("id", "unknown"))
            evidence = (
                fault.get("evidence", "")
                or fault.get("grounding_span", "")
                or "(no evidence available)"
            )
            rule = fault.get("rule_citation", fault.get("rule", "")) or "Rule not cited"
            family = fault.get("failure_family", "unknown")
            expected = "KEEP" if fid in matched_gt_ids else "DOWNGRADE"
            samples.append(ProbeSample(fid, evidence[:500], rule[:200], family, expected))
    return samples


def run_probe_batch(
    samples: list[ProbeSample],
    thinking_mode: Literal["on", "off"],
) -> dict[str, int]:
    """Run the probe batch; return counters for ConformanceRate + DiscriminationAccuracy."""
    from llm.client import get_client, chat_completion_tools
    from llm.reliability import coerce_and_validate, supports_guided_json, build_guided_extra_body
    from schemas.llm import VERDICT, ParseFailed

    client = get_client()
    from config import get_settings
    model = get_settings().verifier_model
    system_prompt = THINKING_ON_SYSTEM if thinking_mode == "on" else THINKING_OFF_SYSTEM
    temperature = 0.6 if thinking_mode == "on" else 0.0
    max_tokens = 2048 if thinking_mode == "on" else 256

    extra = build_guided_extra_body(VERDICT) if supports_guided_json(client, model) else {}

    VERDICT_TOOL = {
        "type": "function",
        "function": {
            "name": "emit_verdict",
            "description": "Emit VERDICT on a candidate deficiency",
            "parameters": {
                "type": "object",
                "properties": {
                    "verdict": {"type": "string", "enum": ["KEEP", "DOWNGRADE"]},
                    "confidence": {"type": "number"},
                    "rationale": {"type": "string"},
                    "grounding_span": {"type": "string"},
                },
                "required": ["verdict", "confidence", "rationale", "grounding_span"],
                "additionalProperties": False,
            }
        }
    }

    counters = {
        "total": 0, "parsed": 0, "parse_failed": 0,
        "correct": 0, "incorrect": 0,
        "keep_count": 0, "downgrade_count": 0,
        "keep_expected_total": 0, "downgrade_expected_total": 0,
    }

    for sample in samples:
        counters["total"] += 1
        user_msg = (
            f"Candidate finding: {sample.finding_id}\n"
            f"Failure family: {sample.failure_family}\n"
            f"Evidence span: {sample.evidence}\n"
            f"Rule violated: {sample.rule_citation}\n\n"
            f"Emit VERDICT: KEEP if this is a real deficiency, DOWNGRADE if it is a false positive."
        )
        turn = chat_completion_tools(
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_msg},
            ],
            tools=[VERDICT_TOOL],
            model=model,
            temperature=temperature,
            max_tokens=max_tokens,
            extra_body=extra or None,
        )

        raw_args = {}
        if turn.tool_calls:
            try:
                raw_args = json.loads(turn.tool_calls[0].function.arguments)
            except Exception:
                pass

        verdict_instance, failure = coerce_and_validate(raw_args, VERDICT)
        # D-12: VERDICT XOR ParseFailed invariant — assert per iteration (W2 fix)
        assert not (verdict_instance is not None and failure is not None), (
            f"D-12 XOR violated on {sample.finding_id}: "
            f"coerce_and_validate returned both a verdict instance AND a ParseFailed — "
            f"exactly one must be non-None. verdict={verdict_instance}, failure={failure}"
        )
        if verdict_instance is not None:
            counters["parsed"] += 1
            emitted = verdict_instance.verdict.value
            counters["keep_count"] += (emitted == "KEEP")
            counters["downgrade_count"] += (emitted == "DOWNGRADE")
            if sample.expected_verdict == "KEEP":
                counters["keep_expected_total"] += 1
                counters["correct"] += (emitted == "KEEP")
            else:
                counters["downgrade_expected_total"] += 1
                counters["correct"] += (emitted == "DOWNGRADE")
        else:
            counters["parse_failed"] += 1

        log.debug(
            "probe_result",
            finding_id=sample.finding_id,
            expected=sample.expected_verdict,
            parsed=(verdict_instance is not None),
            verdict=(verdict_instance.verdict.value if verdict_instance else None),
            mode=thinking_mode,
        )

    return counters


class ConformanceRateMetric(BaseMetric):
    """D-06a: ≥98% machine-parsable VERDICT post-repair, per thinking mode."""
    threshold = 0.98
    name = "conformance_rate"

    def __init__(self, counters: dict):
        self._counters = counters
        self.score = 0.0

    def measure(self, test_case: LLMTestCase) -> float:
        total = self._counters["total"]
        parsed = self._counters["parsed"]
        self.score = parsed / total if total > 0 else 0.0
        log.info("conformance_rate", score=self.score, parsed=parsed, total=total)
        return self.score

    def is_successful(self) -> bool:
        return self.score >= self.threshold


class DiscriminationAccuracyMetric(BaseMetric):
    """D-06b: ≥80% correct on matched-GT vs FP-tail split.
    
    ALSO enforces the constant-verdict tripwire: majority verdict < 95%
    (a blanket-KEEP model at 100% syntax still fails this).
    """
    threshold = 0.80
    name = "discrimination_accuracy"

    def __init__(self, counters: dict):
        self._counters = counters
        self.score = 0.0

    def measure(self, test_case: LLMTestCase) -> float:
        total = self._counters["total"]
        correct = self._counters["correct"]
        self.score = correct / total if total > 0 else 0.0
        keep_ratio = self._counters["keep_count"] / total if total > 0 else 0.0
        downgrade_ratio = self._counters["downgrade_count"] / total if total > 0 else 0.0
        # Constant-verdict tripwire (D-06b): fail if one verdict ≥95%
        if max(keep_ratio, downgrade_ratio) >= 0.95:
            self.score = 0.0  # constant-verdict model = fail regardless of discrimination score
            log.warning(
                "constant_verdict_tripwire_fired",
                keep_ratio=keep_ratio,
                downgrade_ratio=downgrade_ratio,
            )
        log.info(
            "discrimination_accuracy",
            score=self.score,
            correct=correct,
            total=total,
            keep_ratio=keep_ratio,
            downgrade_ratio=downgrade_ratio,
        )
        return self.score

    def is_successful(self) -> bool:
        return self.score >= self.threshold


@pytest.fixture(scope="module")
def probe_samples():
    return load_probe_samples()


@pytest.mark.parametrize("thinking_mode", ["on", "off"])
def test_verifier_conformance_and_discrimination(thinking_mode, probe_samples):
    """D-06 gate: conformance ≥98% AND discrimination ≥80%, both modes separately.
    
    This is the pre-registered, falsifiable two-dimensional pass bar (D-06a + D-06b).
    A model that passes conformance by answering KEEP to everything fails discrimination
    (the constant-verdict tripwire fires at ≥95% majority verdict).
    """
    counters = run_probe_batch(probe_samples, thinking_mode)

    conformance_metric = ConformanceRateMetric(counters)
    discrimination_metric = DiscriminationAccuracyMetric(counters)

    test_case = LLMTestCase(
        input=f"115 real beta candidates, thinking_mode={thinking_mode}",
        actual_output=f"parsed={counters['parsed']}/{counters['total']}, correct={counters['correct']}",
    )

    assert_test(test_case, [conformance_metric, discrimination_metric])
