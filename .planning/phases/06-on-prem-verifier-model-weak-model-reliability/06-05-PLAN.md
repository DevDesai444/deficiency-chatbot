```python
#!/usr/bin/env python3
"""Deploy Nemotron-Super-49B-v1.5 via self-managed vLLM to Databricks GPU_XLARGE_8.

D-03: self-managed vLLM (NOT Databricks managed Provisioned Throughput).
D-04: always-warm, scale_to_zero_enabled=False.
D-20: quant/GPU confirmed in ADR-nemotron-verifier-model.md D-19/D-20 Amendment.
Pattern: mirrors notebooks/deploy_adapters.py (lines 40-145).
"""
from __future__ import annotations

import argparse
import os
import time

import httpx
import mlflow
import mlflow.pyfunc

# === Configuration ===
NEMOTRON_VOLUME_DIR = "/Volumes/defpredict/main/artifacts/nemotron-49b"
UC_MODEL_NAME = "defpredict.main.defpredict_nemotron"
ENDPOINT_NAME = "defpredict-nemotron"
SERVED_MODEL_NAME = "nemotron-super-49b-v1_5"  # MUST match config.verifier_model + _guided_cache key

# === D-19/D-20 confirmed values (read from ADR-nemotron-verifier-model.md D-19/D-20 Amendment) ===
# Update these from the ADR Amendment before running.
TENSOR_PARALLEL_SIZE = 8       # 8×H100 (GPU_XLARGE_8 Hopper working assumption)
DTYPE_FLAG = "fp8"             # FP8 (Hopper default) — or "bfloat16" if FP8 unavailable
# If Blackwell confirmed: TENSOR_PARALLEL_SIZE=1, DTYPE_FLAG="nvfp4" (different flags)

VLLM_CMD = (
    "python3 -m vllm.entrypoints.openai.api_server "
    f"--model {NEMOTRON_VOLUME_DIR} "
    f"--served-model-name {SERVED_MODEL_NAME} "
    "--trust-remote-code "
    f"--tensor-parallel-size {TENSOR_PARALLEL_SIZE} "
    "--max-model-len 65536 "
    "--gpu-memory-utilization 0.95 "
    "--enable-auto-tool-choice "
    f"--tool-parser-plugin {NEMOTRON_VOLUME_DIR}/llama_nemotron_toolcall_parser_no_streaming.py "
    "--tool-call-parser llama_nemotron_json "
    "--host 0.0.0.0 "
    "--port 8080 "
    f"--dtype {DTYPE_FLAG} "
    f"--quantization {DTYPE_FLAG}"
    # structured outputs ON by default (xgrammar/auto backend) — no extra flag
)


class NemotronServingModel(mlflow.pyfunc.PythonModel):
    """Placeholder pyfunc — actual serving done by the vLLM entrypoint baked into metadata."""
    def predict(self, context, model_input):
        raise NotImplementedError("Served via vLLM entrypoint, not pyfunc predict")


def register_model() -> str:
    """Register Nemotron as an MLflow pyfunc in Unity Catalog. Returns model version."""
    mlflow.set_registry_uri("databricks-uc")
    with mlflow.start_run(run_name="register-nemotron"):
        mlflow.pyfunc.log_model(
            name="defpredict_nemotron",
            python_model=NemotronServingModel(),
            artifacts={"model_dir": NEMOTRON_VOLUME_DIR},
            metadata={
                "task": "llm/v1/chat",
                "entrypoint": VLLM_CMD,
            },
            registered_model_name=UC_MODEL_NAME,
            extra_pip_requirements=["vllm>=0.11"],  # ensure correct vLLM on serving container
        )
    print(f"Registered: {UC_MODEL_NAME}")
    return "1"


def create_endpoint(entity_version: str = "1") -> None:
    """Create or update the Databricks GPU_XLARGE_8 serving endpoint."""
    host = os.environ.get("DATABRICKS_HOST", "")
    token = os.environ.get("DATABRICKS_TOKEN", "")
    if not host or not token:
        raise RuntimeError("DATABRICKS_HOST and DATABRICKS_TOKEN must be set in .env")
    headers = {"Authorization": f"Bearer {token}"}
    payload = {
        "name": ENDPOINT_NAME,
        "config": {
            "served_entities": [{
                "entity_name": UC_MODEL_NAME,
                "entity_version": entity_version,
                "workload_type": "GPU_XLARGE_8",  # proven tier on aip-amn-dev (D-04)
                "workload_size": "Small",
                "scale_to_zero_enabled": False,    # D-04: always-warm
            }]
        }
    }
    r = httpx.post(f"{host}/api/2.0/serving-endpoints", headers=headers,
                   json=payload, timeout=30.0)
    if "already exists" in r.text.lower():
        print(f"Endpoint {ENDPOINT_NAME!r} exists — updating config")
        r2 = httpx.put(f"{host}/api/2.0/serving-endpoints/{ENDPOINT_NAME}/config",
                       headers=headers, json=payload["config"], timeout=30.0)
        r2.raise_for_status()
    else:
        r.raise_for_status()
    print(f"Endpoint {ENDPOINT_NAME!r} create/update accepted — waiting for READY")


def wait_for_ready(timeout_minutes: int = 30) -> None:
    """Poll until the endpoint reaches READY state."""
    host = os.environ.get("DATABRICKS_HOST", "")
    token = os.environ.get("DATABRICKS_TOKEN", "")
    headers = {"Authorization": f"Bearer {token}"}
    deadline = time.time() + timeout_minutes * 60
    while time.time() < deadline:
        r = httpx.get(f"{host}/api/2.0/serving-endpoints/{ENDPOINT_NAME}",
                      headers=headers, timeout=15.0)
        state = r.json().get("state", {}).get("ready", "NOT_READY")
        print(f"  Endpoint state: {state}", end="\r")
        if state == "READY":
            print(f"\nEndpoint {ENDPOINT_NAME!r} is READY")
            return
        time.sleep(30)
    raise TimeoutError(f"Endpoint not READY after {timeout_minutes} minutes")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Deploy Nemotron-Super-49B to Databricks")
    parser.add_argument("--register-only", action="store_true",
                        help="Register in UC without creating the serving endpoint")
    parser.add_argument("--wait", action="store_true", default=True,
                        help="Wait for endpoint READY after creation")
    args = parser.parse_args()
    version = register_model()
    if not args.register_only:
        create_endpoint(entity_version=version)
        if args.wait:
            wait_for_ready()
```

"""MODEL-02 pre-wiring probes for the live Nemotron-Super-49B-v1.5 endpoint.

D-18: Thinking ON vs OFF produces different completion_token counts.

from __future__ import annotations

import time
import pytest
import structlog

log = structlog.get_logger()

pytestmark = pytest.mark.integration

# D-18: Reasoning toggle strings (from NVIDIA model card v1.5)
# PROBE BOTH — compare completion_tokens; higher count = <think> block = ON
THINKING_ON_SYSTEM = (
    "You are a regulatory compliance verifier. detailed thinking on\n"
    "Analyze the candidate deficiency carefully before emitting the VERDICT."
)
THINKING_OFF_SYSTEM = (
    "/no_think\n"
    "You are a regulatory compliance verifier. Emit the VERDICT directly."
)

# Minimal VERDICT tool schema for the round-trip probe
VERDICT_TOOL = {
    "type": "function",
    "function": {
        "name": "emit_verdict",
        "description": "Emit VERDICT on a candidate deficiency",
        "parameters": {
            "type": "object",
            "properties": {
                "verdict": {"type": "string", "enum": ["KEEP", "DOWNGRADE"]},
                "confidence": {"type": "number"},
                "rationale": {"type": "string"},
                "grounding_span": {"type": "string"},
            },
            "required": ["verdict", "confidence", "rationale", "grounding_span"],
            "additionalProperties": False,
        }
    }
}

PROBE_MESSAGES = [
    {
        "role": "user",
        "content": (
            "Candidate: The stability study protocol omits the proposed shelf-life "
            "justification required by ICH Q1A(R2) §3.4.\n"
            "Source span: 'Stability studies were conducted at 25°C/60% RH.'\n"
            "Rule: ICH Q1A(R2) §3.4 — shelf-life justification required.\n"
            "Emit VERDICT."
        ),
    }
]

def test_nemotron_tool_call_returns_structured_tool_calls(nemotron_client, nemotron_model):
    """Pitfall 1 guard: tool_calls populated, content empty/None.
    
    If wrong --tool-call-parser: tool_calls=[], content=response text.
    If correct parser (llama_nemotron_json): tool_calls=[...], content=None or "".
    """
    from llm.client import chat_completion_tools
    from llm.reliability import supports_guided_json, build_guided_extra_body

    extra = {}
    if supports_guided_json(nemotron_client, nemotron_model):
        extra = build_guided_extra_body(__import__("schemas.llm", fromlist=["VERDICT"]).VERDICT)

    turn = chat_completion_tools(
        messages=[{"role": "system", "content": THINKING_OFF_SYSTEM}] + PROBE_MESSAGES,
        tools=[VERDICT_TOOL],
        model=nemotron_model,
        temperature=0.0,
        max_tokens=256,
        extra_body=extra or None,
    )
    assert turn.tool_calls, (
        f"tool_calls is empty — wrong --tool-call-parser or model responded with prose. "
        f"content={turn.content!r}"
    )
    assert not turn.content or turn.content.strip() == "", (
        f"content is non-empty — prose leaked from tool-call turn. content={turn.content!r[:200]}"
    )


@pytest.mark.parametrize("mode,system_prompt,temperature,max_tokens", [
    ("on", THINKING_ON_SYSTEM, 0.6, 2048),
    ("off", THINKING_OFF_SYSTEM, 0.0, 256),
])
def test_thinking_on_inflates_completion_tokens(
    nemotron_client, nemotron_model, mode, system_prompt, temperature, max_tokens
):
    """Pitfall 2 guard: D-18 — ON produces more tokens than OFF due to <think> block.
    
    If the toggle string is wrong: both modes produce identical token counts.
    Record both counts via structlog for Phase 7 escalation-trigger design.
    """
    from llm.client import chat_completion_tools

    t0 = time.perf_counter()
    turn = chat_completion_tools(
        messages=[{"role": "system", "content": system_prompt}] + PROBE_MESSAGES,
        tools=[VERDICT_TOOL],
        model=nemotron_model,
        temperature=temperature,
        max_tokens=max_tokens,
    )
    wall_ms = (time.perf_counter() - t0) * 1000

    completion_tokens = (turn.usage.completion_tokens if turn.usage else None)
    prompt_tokens = (turn.usage.prompt_tokens if turn.usage else None)
    log.info(
        "d18_thinking_mode_probe",
        mode=mode,
        wall_ms=round(wall_ms, 1),
        prompt_tokens=prompt_tokens,
        completion_tokens=completion_tokens,
        usage_present=(turn.usage is not None),
    )

    # Soft assertion: ON should produce more tokens than OFF.
    # This test is parametrized — compare via a module-level dict keyed by mode.
    # The hard assertion (ON > OFF) runs in test_thinking_mode_token_split_is_detected.
    assert completion_tokens is not None or True, "usage missing — flag for D-18 estimate"


def test_thinking_mode_token_split_is_detected(nemotron_client, nemotron_model):
    """D-18: Run both modes and assert ON completion_tokens > OFF completion_tokens.
    
    This is the definitive check that the reasoning toggle string is correct.
    If both produce equal token counts, the toggle is ineffective.
    """
    from llm.client import chat_completion_tools
    results = {}
    for mode, system_prompt, temperature, max_tokens in [
        ("on", THINKING_ON_SYSTEM, 0.6, 2048),
        ("off", THINKING_OFF_SYSTEM, 0.0, 256),
    ]:
        turn = chat_completion_tools(
            messages=[{"role": "system", "content": system_prompt}] + PROBE_MESSAGES,
            tools=[VERDICT_TOOL],
            model=nemotron_model,
            temperature=temperature,
            max_tokens=max_tokens,
        )
        results[mode] = turn.usage.completion_tokens if turn.usage else 0

    assert results["on"] > results["off"], (
        f"Thinking toggle may be ineffective: ON tokens={results['on']}, OFF tokens={results['off']}. "
        f"Probe both toggle strings ('/no_think' vs 'detailed thinking off') per Pitfall 2."
    )
