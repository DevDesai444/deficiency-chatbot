```python
def test_threshold_transfer_structural_fixture_b(tmp_path):
    """THRESHOLD-TRANSFER: frozen threshold fires on DIFFERENT data (different surface forms).
    fixture_b uses dissolution domain vocabulary, different compound names and numeric values
    than fixture_a — but the SAME structural violation TYPES (labeled-aggregate recompute).
    This is the anti-overfitting witness: if it only passed with fixture_a renamed, it proves
    nothing about transfer. Fixture_b proves the threshold generalizes to new data."""
    assert _SYNTHETIC_FIXTURE_B_DIR.exists(), (
        "fixture_b must be committed at tests/fixtures/synthetic_submission_b/ (B3 fix)")
    assert _SYNTHETIC_FIXTURE_DIR.exists(), "fixture_a must be committed (D-GRD1)"
    # Run structural on BOTH fixtures with the SAME threshold (frozen from fixture_a measurement)
    import json, pathlib
    threshold_path = pathlib.Path("src/evals/baseline/structural_threshold.json")
    threshold = json.loads(threshold_path.read_text())["threshold"]

    corpus_a = ingest_corpus(_SYNTHETIC_FIXTURE_DIR, cache_dir=str(tmp_path / "cache_a"))
    ledger_a = RetrievalLedger()
    faults_a = detect_structural_inconsistencies(corpus_a, corpus_a.manifest, ledger_a)
    struct_a = [f for f in faults_a if f.structural_anchor]

    corpus_b = ingest_corpus(_SYNTHETIC_FIXTURE_B_DIR, cache_dir=str(tmp_path / "cache_b"))
    ledger_b = RetrievalLedger()
    faults_b = detect_structural_inconsistencies(corpus_b, corpus_b.manifest, ledger_b)
    struct_b = [f for f in faults_b if f.structural_anchor]

    # Both fixtures must produce structural violations (same threshold, different data -> transfer)
    assert len(struct_a) > 0, "fixture_a: no structural violations found (SAME-LOGIC failed)"
```

```python
def cmd_deterministic_recall_gate(args) -> int:
    """Combined gate for all deterministic recall legs on the synthetic fixture (Phase 5 SC5)."""
    from ingest.corpus import ingest_corpus
    from rulebook.structural import detect_structural_inconsistencies
    from rulebook.references import extract_references, detect_reference_anomalies
    from rulebook.precedent_search import detect_precedent_candidates
    from tools.ledger import RetrievalLedger
    import pathlib, tempfile
    fixture = pathlib.Path("src/evals/dataset/synthetic_fixture")
    if not fixture.exists():
        print("ERROR: synthetic fixture missing"); return 1
    with tempfile.TemporaryDirectory() as tmp:
        corpus = ingest_corpus(fixture, cache_dir=tmp + "/cache")
        # Structural leg
        ledger = RetrievalLedger()
        struct_faults = detect_structural_inconsistencies(corpus, corpus.manifest, ledger)
        # Reference leg
        extract_references(corpus, corpus.manifest, db_path=tmp + "/edges.db")
        ledger_ref = RetrievalLedger()
        ref_faults = detect_reference_anomalies(corpus, corpus.manifest, ledger_ref, db_path=tmp + "/edges.db")
        # Precedent leg (report but don't hard-fail — needs real embeddings)
        ledger_prec = RetrievalLedger()
        try:
            prec_faults = detect_precedent_candidates(corpus, corpus.manifest, ledger_prec)
        except Exception as e:
            prec_faults = []
            print(f"  precedent-leg skipped (no embedding backend?): {e}")
    print(f"  structural: {len(struct_faults)} finding(s) with structural_anchor")
    print(f"  reference:  {len(ref_faults)} finding(s) with reference_anchor")
    print(f"  precedent:  {len(prec_faults)} finding(s) with precedent_anchor")
    if not struct_faults:
        print("FAIL: structural leg found no findings on synthetic fixture"); return 1
    if not ref_faults:
        print("FAIL: reference leg found no findings on synthetic fixture"); return 1
    print("PASS: deterministic-recall-gate")
    return 0
```
