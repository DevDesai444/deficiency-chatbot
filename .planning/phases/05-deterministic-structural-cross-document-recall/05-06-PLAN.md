```python
def cmd_deterministic_recall_gate(args) -> int:
    """Combined gate for all deterministic recall legs on the synthetic fixture (Phase 5 SC5)."""
    from ingest.corpus import ingest_corpus
    from rulebook.structural import detect_structural_inconsistencies
    from rulebook.references import extract_references, detect_reference_anomalies
    from rulebook.precedent_search import detect_precedent_candidates
    from tools.ledger import RetrievalLedger
    import pathlib, tempfile
    fixture = pathlib.Path("src/evals/dataset/synthetic_fixture")
    if not fixture.exists():
        print("ERROR: synthetic fixture missing"); return 1
    with tempfile.TemporaryDirectory() as tmp:
        corpus = ingest_corpus(fixture, cache_dir=tmp + "/cache")
        # Structural leg
        ledger = RetrievalLedger()
        struct_faults = detect_structural_inconsistencies(corpus, corpus.manifest, ledger)
        # Reference leg
        extract_references(corpus, corpus.manifest, db_path=tmp + "/edges.db")
        ledger_ref = RetrievalLedger()
        ref_faults = detect_reference_anomalies(corpus, corpus.manifest, ledger_ref, db_path=tmp + "/edges.db")
        # Precedent leg (report but don't hard-fail — needs real embeddings)
        ledger_prec = RetrievalLedger()
        try:
            prec_faults = detect_precedent_candidates(corpus, corpus.manifest, ledger_prec)
        except Exception as e:
            prec_faults = []
            print(f"  precedent-leg skipped (no embedding backend?): {e}")
    print(f"  structural: {len(struct_faults)} finding(s) with structural_anchor")
    print(f"  reference:  {len(ref_faults)} finding(s) with reference_anchor")
    print(f"  precedent:  {len(prec_faults)} finding(s) with precedent_anchor")
    if not struct_faults:
        print("FAIL: structural leg found no findings on synthetic fixture"); return 1
    if not ref_faults:
        print("FAIL: reference leg found no findings on synthetic fixture"); return 1
    print("PASS: deterministic-recall-gate")
    return 0
```
