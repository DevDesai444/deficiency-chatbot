```python
def build_doc_index(cache_dir: str, key: str, nt: NormalizedText, doc_id: str) -> None:
    """Build and persist per-doc search index sidecars. Called once at ingest time.
    
    Ruling 9 (.npy atomic write): uses open(tmp, 'wb') file handle for np.save
    to avoid np.save's .npy-appending behavior on paths not ending in .npy.
    """
    import os, json
    from pathlib import Path
    try:
        chunks = _build_chunks(nt, doc_id)  # same chunking as existing search_corpus
        if not chunks:
            return
        chunk_texts = [c["text"] for c in chunks]
        embeddings = embed_texts(chunk_texts)  # shape: (n_chunks, dim)
        tokenized = [t.lower().split() for t in chunk_texts]
        
        cache = Path(cache_dir)
        # .chunks.json: atomic text write
        chunks_path = cache / f"{key}.chunks.json"
        chunks_tmp = cache / f"{key}.chunks.json.tmp"
        chunks_tmp.write_text(json.dumps(chunks), encoding="utf-8")
        os.replace(chunks_tmp, chunks_path)
        
        # .bm25.json: atomic text write
        bm25_path = cache / f"{key}.bm25.json"
        bm25_tmp = cache / f"{key}.bm25.json.tmp"
        bm25_tmp.write_text(json.dumps(tokenized), encoding="utf-8")
        os.replace(bm25_tmp, bm25_path)
        
        # .embeddings.npy: CORRECT atomic write (Ruling 9 — use open file handle)
        emb_path = cache / f"{key}.embeddings.npy"
        emb_tmp = cache / f"{key}.embeddings.npy.tmp"
        with open(emb_tmp, "wb") as f:
            np.save(f, embeddings.astype("float32"))
        os.replace(emb_tmp, emb_path)
        
    except Exception as e:
        import structlog
        structlog.get_logger().warning("build_doc_index_failed", doc_id=doc_id, error=str(e))
```
