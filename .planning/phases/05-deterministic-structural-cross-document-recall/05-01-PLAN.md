```python
import zipfile, xml.etree.ElementTree as ET
_HYPERLINK_TYPE = "http://schemas.openxmlformats.org/officeDocument/2006/relationships/hyperlink"
_RELS_NS = "http://schemas.openxmlformats.org/package/2006/relationships"

def _extract_docx_hyperlinks(path: str) -> list[dict]:
    try:
        with zipfile.ZipFile(path) as z:
            if "word/_rels/document.xml.rels" not in z.namelist():
                return []
            rels_xml = z.read("word/_rels/document.xml.rels")
        root = ET.fromstring(rels_xml)
        links = []
        for rel in root.findall(f"{{{_RELS_NS}}}Relationship"):
            if rel.get("Type") == _HYPERLINK_TYPE:
                links.append({"rId": rel.get("Id", ""), "target": rel.get("Target", ""), "paragraph_index": None})
                if len(links) >= 1000:
                    import logging; logging.getLogger(__name__).warning("docx hyperlink cap reached at 1000 entries")
                    break
        return links
    except Exception:
        return []
```
